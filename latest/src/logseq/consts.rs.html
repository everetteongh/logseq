40/// Regular expression to parse [`Due`] Logseq properties. This follows standard [Org Mode format](https://orgmode.org/manual/Deadlines-and-Scheduling.html), so might be delegated to an Org dependency in the future.
41#[allow(clippy::unwrap_used)]
42pub static DUE_REGEX: LazyLock<Regex> = LazyLock::new(|| {
43    Regex::new(
44        r"(SCHEDULED|DEADLINE):\s*<(\d{4}-\d{2}-\d{2})\s+([A-Za-z]{3})(?:\s+(\d{1,2}:\d{2}))?(?:\s+([\.\+]*\+\d+[ymwdh]))?>$"
45    ).unwrap()
46});
