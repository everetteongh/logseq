1use crate::{
2    consts::LOGSEQ_EXCLUDE,
3    error::{GraphBuilderError, Logseq},
4    graph::Graph,
5};
6use std::path::PathBuf;

8/// A helper object for constructing a [`Graph`].
9pub struct GraphBuilder {
10    /// Paths for the [`WalkDir`] directory crawler to exclude.
11    exclude: Vec<String>,
12    /// The root of the Logseq graph.
13    dir: Option<PathBuf>,
14}
15
16impl Default for GraphBuilder {
17    fn default() -> Self {
18        Self {
19            exclude: LOGSEQ_EXCLUDE.map(String::from).to_vec(),
20            dir: None,
21        }
22    }
23}
24
25impl GraphBuilder {
26    /// Paths for the [`walkdir::WalkDir`] directory crawler to exclude.
27    #[must_use]
28    pub fn exclude(mut self, exclude: &[String]) -> Self {
29        self.exclude = exclude.to_vec();
30        self
31    }
32    /// The root of the Logseq graph.
33    #[must_use]
34    pub fn dir(mut self, dir: PathBuf) -> Self {
35        self.dir = Some(dir);
36        self
37    }
38    /// Try to construct a [`Graph`] object from the builder fields.
39    ///
40    /// # Errors
41    /// Fails if the root directory (`self.dir`) is unset.
42    pub fn build(self) -> Result<Graph, Logseq> {
43        let dir = self.dir.ok_or(GraphBuilderError::UndefinedRootDirectory)?;
44
45        Ok(Graph {
46            exclude: self.exclude,
47            dir,
48        })
49    }
50}
