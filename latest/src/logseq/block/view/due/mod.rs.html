23/// Parser for SCHEDULED/DEADLINE. This follows standard [Org Mode format](https://orgmode.org/manual/Deadlines-and-Scheduling.html), so might be delegated to an Org dependency in the future.
24#[derive(Debug, Clone)]
25#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
26pub struct Due {
27    /// The due date.
28    pub date: Date,
29    /// The due day.
30    pub day: Weekday,
31    /// The due time.
32    pub time: Option<Time>,
33    /// The due repeater.
34    pub repeater: Option<DueRepeater>,
35    /// The due kind; one of [`DueKind::Scheduled`], [`DueKind::Deadline`].
36    pub kind: DueKind,
37}
38
39impl fmt::Display for Due {
40    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
41        // NOTE: `DATE_FORMAT` is guaranteed valid @ compile time, so **this will never panic**.
42        #[allow(clippy::unwrap_used)]
43        write!(
44            f,
45            "{} <{} {}",
46            self.kind,
47            self.date.format(DATE_FORMAT).unwrap(),
48            shorten_weekday(&self.day.to_string())
49        )?;
50
51        if let Some(time) = &self.time {
52            // NOTE: `TIME_FORMAT` is guaranteed valid @ compile time, so **this will never panic**.
53            #[allow(clippy::unwrap_used)]
54            write!(f, " {}", time.format(TIME_FORMAT).unwrap())?;
55        }
56
57        if let Some(repeater) = &self.repeater {
58            write!(f, " {repeater}")?;
59        }
60
61        write!(f, ">")
62    }
63}

65impl FromStr for Due {
66    type Err = Logseq;
67
68    fn from_str(s: &str) -> Result<Self, Self::Err> {
69        if let Some(captures) = DUE_REGEX.captures(s) {
70            let kind = captures
71                .get(1)
72                .ok_or(ParseDueError::InvalidInput)
73                .and_then(|m| DueKind::from_str(m.as_str()))?;
74            let date = captures
75                .get(2)
76                .ok_or(Logseq::ParseScheduled(ParseDueError::InvalidInput))
77                .and_then(|m| Date::parse(m.as_str(), DATE_FORMAT).map_err(Logseq::from))?;
78            let day = captures
79                .get(3)
80                .ok_or(Logseq::ParseScheduled(ParseDueError::InvalidInput))
81                .and_then(|m| custom_parse_weekday(m.as_str()).map_err(Logseq::from))?;
82            let time = captures
83                .get(4)
84                .map(|m| m.as_str())
85                .and_then(|t| Time::parse(t, TIME_FORMAT).ok());
86            let repeater = captures
87                .get(5)
88                .map(|m| m.as_str())
89                .and_then(|r| DueRepeater::from_str(r).ok());
90
91            return Ok(Self {
92                date,
93                day,
94                time,
95                repeater,
96                kind,
97            });
98        }
99
100        Err(ParseDueError::InvalidInput.into())
101    }
102}

104/// Parse strings as weekdays with support for shortened weekdays.
105fn custom_parse_weekday(s: &str) -> Result<Weekday, InvalidVariant> {
106    match s.trim().to_lowercase().as_str() {
107        "mon" | "monday" => Ok(Weekday::Monday),
108        "tue" | "tuesday" => Ok(Weekday::Tuesday),
109        "wed" | "wednesday" => Ok(Weekday::Wednesday),
110        "thu" | "thursday" => Ok(Weekday::Thursday),
111        "fri" | "friday" => Ok(Weekday::Friday),
112        "sat" | "saturday" => Ok(Weekday::Saturday),
113        "sun" | "sunday" => Ok(Weekday::Sunday),
114        _ => Err(InvalidVariant),
115    }
116}
117
118/// Convert a weekday string into its shortened counterpart.
119fn shorten_weekday(s: &str) -> &str {
120    match s.trim() {
121        "Monday" => "Mon",
122        "Tuesday" => "Tue",
123        "Wednesday" => "Wed",
124        "Thursday" => "Thu",
125        "Friday" => "Fri",
126        "Saturday" => "Sat",
127        "Sunday" => "Sun",
128        other => other,
129    }
130}

132#[cfg(feature = "icalendar")]
133#[allow(clippy::fallible_impl_from)]
134impl From<Due> for DatePerhapsTime {
135    fn from(due: Due) -> Self {
136        // NOTE: We already know our date data is valid because we have a parsed object from `time`, so **this will never panic.**
137        #[allow(clippy::unwrap_used)]
138        let date_naive = NaiveDate::from_ymd_opt(
139            due.date.year(),
140            due.date.month() as u32,
141            due.date.day().into(),
142        )
143        .unwrap();
144
145        // NOTE: We already know our date data is valid because we have a parsed object from `time`, so **this will never panic.**
146        #[allow(clippy::unwrap_used)]
147        due.time.map_or_else(
148            || date_naive.into(),
149            |time| {
150                CalendarDateTime::Floating(
151                    date_naive
152                        .and_hms_opt(
153                            time.hour().into(),
154                            time.minute().into(),
155                            time.second().into(),
156                        )
157                        .unwrap(),
158                )
159                .into()
160            },
161        )
162    }
