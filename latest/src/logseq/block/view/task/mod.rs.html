16/// View over a [`Block`] which starts with a [`TaskStatus`] marker..
17#[derive(Debug, Clone)]
18pub struct Task<'a> {
19    /// Underlying block.
20    pub block: &'a Block,
21    /// The task's status.
22    pub status: TaskStatus,
23    /// The task's label.
24    pub label: String,
25    /// The task's priority.
26    pub priority: Option<TaskPriority>,
27    /// The task's due date.
28    pub due: Option<Due>,
29}
30
31/// Same as [`Task`], but mutable.
32#[derive(Debug)]
33pub struct TaskMut<'a> {
34    /// Underlying mutable block.
35    pub block: &'a mut Block,
36    /// The task's status.
37    pub status: TaskStatus,
38    /// The task's label.
39    pub label: String,
40    /// The task's priority.
41    pub priority: Option<TaskPriority>,
42    /// The task's due date.
43    pub due: Option<Due>,
44}
45
46impl TaskMut<'_> {
47    /// Update the status of a task.
48    pub fn status(&mut self, status: &TaskStatus) {
49        self.block.markdown =
50            self.block
51                .markdown
52                .replacen(&self.status.to_string(), &status.to_string(), 1);
53        self.status = status.clone();
54    }
55}
56
57impl<'a> TryFrom<&'a mut Block> for TaskMut<'a> {
58    type Error = TaskError;
59
60    fn try_from(block: &'a mut Block) -> Result<Self, Self::Error> {
61        let plain = block.content();
62        let mut words = plain.split_whitespace().peekable();
63
64        let status = TaskStatus::from_str(words.next().ok_or(TaskError::EmptyItem)?)?;
65        let priority = words.peek().and_then(|w| w.parse().ok()).inspect(|_| {
66            words.next();
67        });
68
69        let label = words.collect::<Vec<_>>().join(" ");
70        let (label, maybe_due_str) = DUE_DELIMS
71            .iter()
72            .find_map(|d| label.find(d))
73            .map_or((label.as_str(), ""), |idx| label.split_at(idx));
74
75        Ok(Self {
76            label: label.to_string(),
77            due: Due::from_str(maybe_due_str).ok(),
78            block,
79            status,
80            priority,
81        })
82    }
83}
