1use rustc_hash::FxHashMap;
2use std::fmt;
3use uuid::Uuid;
4
5use crate::error::BlockError;

7/// Block properties. See [the official Logseq documentation](https://github.com/logseq/docs/blob/08f855f24d66e4509b7ea808554c13b4649e6ee1/pages/Built-in%20Properties.md).
8#[derive(Debug, Clone)]
9pub struct BlockProperties {
10    /// Designates a page/block as a template.
11    pub template: Option<String>,
12    /// Specifies whether the parent level content of a block should be included when using a template.
13    pub template_including_parent: Option<bool>,
14    /// Whether or not this block is collapsed.
15    pub collapsed: bool,
16    /// The unique [`Uuid`] for this block.
17    pub id: Uuid,
18    /// Any other properties.
19    pub custom: FxHashMap<String, String>,
20}
21
22impl Default for BlockProperties {
23    fn default() -> Self {
24        Self {
25            template: None,
26            template_including_parent: None,
27            collapsed: false,
28            id: Uuid::new_v4(),
29            custom: FxHashMap::default(),
30        }
31    }
32}
33
34impl TryFrom<FxHashMap<String, String>> for BlockProperties {
35    type Error = BlockError;
36
37    fn try_from(map: FxHashMap<String, String>) -> Result<Self, Self::Error> {
38        let mut properties = Self::default();
39
40        for (key, value) in map {
41            match key.as_str() {
42                "template" => properties.template = Some(value),
43                "template-including-parent" => {
44                    properties.template_including_parent = Some(value.trim() == "true");
45                }
46                "collapsed" => properties.collapsed = value.trim() == "true",
47                "id" => {
48                    properties.id =
49                        Uuid::try_parse(&value).map_err(|_| BlockError::InvalidID(value))?;
50                }
51                _ => _ = properties.custom.insert(key, value),
52            }
53        }
54
55        Ok(properties)
56    }
57}
58
59impl fmt::Display for BlockProperties {
60    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
61        if let Some(template) = &self.template {
62            writeln!(f, "template:: {template}")?;
63            if self.template_including_parent.unwrap_or(false) {
64                writeln!(f, "template-including-parent:: true")?;
65            }
66        }
67
68        if self.collapsed {
69            writeln!(f, "collapsed:: true")?;
70        }
71
72        for (key, value) in &self.custom {
73            writeln!(f, "{key}:: {value}")?;
74        }
75
76        write!(f, "id:: {}", self.id)?;
77
78        Ok(())
79    }
